import requests API_KEY = "fmpay_tumhari_key_yahan" API_URL = "https://fampay-verify-api.onrender.com" def verify_payment(utr, amount): res = requests.post( f"{API_URL}/verify", json={ "api_key": API_KEY, "utr": utr, "amount": amount }, timeout=30 ) data = res.json() if data.get("verified"): return True # ✅ Balance add karo elif "already used" in str(data.get("error", "")): return "duplicate" # ❌ UTR reuse attempt else: return False # ❌ Not found
const axios = require('axios'); const API_KEY = 'fmpay_tumhari_key_yahan'; const API_URL = 'https://fampay-verify-api.onrender.com'; async function verifyPayment(utr, amount) { const res = await axios.post(`${API_URL}/verify`, { api_key: API_KEY, utr: utr, amount: amount }); const data = res.data; if (data.verified) return 'verified'; // ✅ if (data.error?.includes('already used')) return 'duplicate'; // ❌ return 'failed'; // ❌ }
